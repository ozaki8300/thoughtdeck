"use client";

import { useRouter } from "next/navigation";
import { useEffect, useState } from "react";

export default function Home() {
  const router = useRouter();
  const [hasDeck, setHasDeck] = useState(false);

  useEffect(() => {
    try {
      const saved = localStorage.getItem("thoughtdeck:mydecks:v1");
      setHasDeck(!!saved);
    } catch {
      setHasDeck(false);
    }
  }, []);

  return (
    <main className="flex min-h-screen items-center justify-center bg-black text-blue-100">
      <div className="flex flex-col items-center gap-6">

        {/* 新規思考 */}
        <button
          onClick={() => router.push("/thoughtdeck")}
          className="rounded-lg border border-blue-400/70 px-6 py-3 hover:bg-blue-500/20 transition"
        >
          ＋ 新しく思考する
        </button>

        {/* MyDeck（データある時だけ） */}
        {hasDeck && (
          <button
            onClick={() => router.push("/mydecks")}
            className="rounded-lg border border-blue-400/70 px-6 py-3 hover:bg-blue-500/20 transition"
          >
            MyDeck
          </button>
        )}

      </div>
    </main>
  );
}
